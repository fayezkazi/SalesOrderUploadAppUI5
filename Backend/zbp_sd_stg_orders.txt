CLASS lhc_zi_sd_stg_orders IMPLEMENTATION.

  METHOD get_instance_authorizations.

  ENDMETHOD.

  METHOD setadmindata.

    READ ENTITIES OF zi_sd_stg_orders IN LOCAL MODE
          ENTITY orders
            FIELDS ( createdat createdby )
            WITH CORRESPONDING #( keys )
          RESULT DATA(lt_orders)
          FAILED DATA(lv_failed).

    DATA lt_update TYPE TABLE FOR UPDATE zi_sd_stg_orders\\orders.
    DATA lv_timestamp TYPE timestampl.

    GET TIME STAMP FIELD lv_timestamp.

    LOOP AT lt_orders INTO DATA(ls_order).
      IF ls_order-createdat IS INITIAL.
        APPEND VALUE #(
            stgguid                    = ls_order-stgguid
            createdat                  = lv_timestamp
            createdby                  = sy-uname "cl_abap_context_info=>get_user_alias( )
            lastchangedat              = lv_timestamp
            lastchangedby              = sy-uname "cl_abap_context_info=>get_user_alias( )
            %control-createdat         = if_abap_behv=>mk-on
            %control-createdby         = if_abap_behv=>mk-on
            %control-lastchangedat     = if_abap_behv=>mk-on
            %control-lastchangedby     = if_abap_behv=>mk-on
        ) TO lt_update.
      ENDIF.
    ENDLOOP.

    IF lt_update IS NOT INITIAL.
      MODIFY ENTITIES OF zi_sd_stg_orders IN LOCAL MODE
        ENTITY orders
          UPDATE FIELDS ( createdat createdby lastchangedat lastchangedby )
          WITH lt_update
        REPORTED DATA(lv_reported).
    ENDIF.

  ENDMETHOD.

  METHOD validatemandatory.

    READ ENTITIES OF zi_sd_stg_orders IN LOCAL MODE
          ENTITY orders
            FIELDS ( auart vkorg matnr )
            WITH CORRESPONDING #( keys )
          RESULT DATA(lt_orders)
          FAILED DATA(lv_failed).

    LOOP AT lt_orders INTO DATA(ls_order).

      IF ls_order-auart IS INITIAL.
        APPEND VALUE #( stgguid = ls_order-stgguid ) TO failed-orders.
        APPEND VALUE #(
            stgguid              = ls_order-stgguid
            %msg                 = new_message_with_text(
                                      severity = if_abap_behv_message=>severity-error
                                      text     = 'Order Type (AUART) is mandatory' )
            %element-auart       = if_abap_behv=>mk-on
        ) TO reported-orders.
      ENDIF.

      IF ls_order-vkorg IS INITIAL.
        APPEND VALUE #( stgguid = ls_order-stgguid ) TO failed-orders.
        APPEND VALUE #(
            stgguid              = ls_order-stgguid
            %msg                 = new_message_with_text(
                                      severity = if_abap_behv_message=>severity-error
                                      text     = 'Sales Organisation (VKORG) is mandatory' )
            %element-vkorg       = if_abap_behv=>mk-on
        ) TO reported-orders.
      ENDIF.

      IF ls_order-matnr IS INITIAL.
        APPEND VALUE #( stgguid = ls_order-stgguid ) TO failed-orders.
        APPEND VALUE #(
            stgguid              = ls_order-stgguid
            %msg                 = new_message_with_text(
                                      severity = if_abap_behv_message=>severity-error
                                      text     = 'Material (MATNR) is mandatory' )
            %element-matnr       = if_abap_behv=>mk-on
        ) TO reported-orders.
      ENDIF.

    ENDLOOP.
  ENDMETHOD.

  METHOD validatesalesorg.

    READ ENTITIES OF zi_sd_stg_orders IN LOCAL MODE
          ENTITY orders
            FIELDS ( vkorg )
            WITH CORRESPONDING #( keys )
          RESULT DATA(lt_orders)
          FAILED DATA(lv_failed).

    LOOP AT lt_orders INTO DATA(ls_order).

      IF ls_order-vkorg IS INITIAL.
        CONTINUE.  " already handled by validateMandatory
      ENDIF.

      " Check TVKO – Sales Organisation master data
      SELECT SINGLE vkorg
        FROM  tvko
        WHERE vkorg = @ls_order-vkorg
        INTO  @DATA(lv_vkorg).

      IF sy-subrc <> 0.
        APPEND VALUE #( stgguid = ls_order-stgguid ) TO failed-orders.
        APPEND VALUE #(
            stgguid              = ls_order-stgguid
            %msg                 = new_message_with_text(
                                      severity = if_abap_behv_message=>severity-error
                                      text     = |Sales Org '{ ls_order-vkorg }' does not exist| )
            %element-vkorg       = if_abap_behv=>mk-on
        ) TO reported-orders.
      ENDIF.

    ENDLOOP.
  ENDMETHOD.

  METHOD excel_upload.

    "── Local types ──────────────────────────────────────────────────────────
    TYPES:
      BEGIN OF ty_order_row,
        vbak_auart      TYPE zsd_stg_orders-auart,
        vbak_vkorg      TYPE zsd_stg_orders-vkorg,
        vbak_vtweg      TYPE zsd_stg_orders-vtweg,
        vbak_spart      TYPE zsd_stg_orders-spart,
        vbak_vkbur      TYPE zsd_stg_orders-vkbur,
        vbak_kunnr      TYPE zsd_stg_orders-kunnr_ag,
        vbrk_kunwe      TYPE zsd_stg_orders-kunnr_we,
        vbak_augru      TYPE zsd_stg_orders-augru,
        vbak_bstnk      TYPE zsd_stg_orders-bstnk,
        vbak_bstdk      TYPE zsd_stg_orders-bstdk,
        vbak_vdatu      TYPE zsd_stg_orders-vdatu,
        kna1_name1      TYPE zsd_stg_orders-name1,
        kna1_name2      TYPE zsd_stg_orders-name2,
        kna1_stras      TYPE zsd_stg_orders-stras,
        adrc_str_suppl1 TYPE zsd_stg_orders-str_suppl,
        kna1_ort01      TYPE zsd_stg_orders-ort01,
        kna1_regio      TYPE zsd_stg_orders-regio,
        kna1_pstlz      TYPE zsd_stg_orders-pstlz,
        adr6_smtp_addr  TYPE zsd_stg_orders-smtp_addr,
        vbap_posnr      TYPE zsd_stg_orders-posnr,
        vbap_matnr      TYPE zsd_stg_orders-matnr,
        vbap_kwmeng     TYPE zsd_stg_orders-kwmeng,
        vbap_meins      TYPE zsd_stg_orders-vrkme,
        vbap_charg      TYPE zsd_stg_orders-charg,
        vbap_werks      TYPE zsd_stg_orders-werks,

      END OF ty_order_row,
      tt_order_rows TYPE STANDARD TABLE OF ty_order_row WITH DEFAULT KEY.

    DATA lt_rows    TYPE tt_order_rows.
    DATA lt_create  TYPE TABLE FOR CREATE zi_sd_stg_orders\\orders.
    DATA lt_created TYPE TABLE FOR READ RESULT zi_sd_stg_orders\\orders.

    "── Guard: action must have exactly one entry with non-empty filecontent ──
    READ TABLE keys INDEX 1 INTO DATA(ls_key).
    IF sy-subrc <> 0 OR ls_key-%param-filecontent IS INITIAL.
      RETURN.
    ENDIF.

    "── Deserialise JSON array ───────────────────────────────────────────────
    " /ui2/cl_json is available in S/4HANA and BTP ABAP Environment.
    " For systems without it, replace with your preferred JSON library.
    TRY.
        /ui2/cl_json=>deserialize(
          EXPORTING json        = ls_key-%param-filecontent
                    pretty_name = /ui2/cl_json=>pretty_mode-camel_case
          CHANGING  data        = lt_rows ).
      CATCH cx_root INTO DATA(lx_json).
        APPEND VALUE #(
            %cid = ls_key-%cid
            %msg = new_message_with_text(
                     severity = if_abap_behv_message=>severity-error
                     text     = |JSON parse error: { lx_json->get_text( ) }| )
        ) TO reported-orders.
        RETURN.
    ENDTRY.

    IF lt_rows IS INITIAL.
      RETURN.
    ENDIF.

    "── Build EML create table ───────────────────────────────────────────────
    DATA(lv_counter) = 1.
    LOOP AT lt_rows INTO DATA(ls_row).
      APPEND VALUE #(
          %cid     = |UPLOAD_{ lv_counter }|
          auart    = ls_row-vbak_auart
          vkorg    = ls_row-vbak_vkorg
          vtweg    = ls_row-vbak_vtweg
          spart    = ls_row-vbak_spart
          vkbur    = ls_row-vbak_vkbur
          kunnrag  = ls_row-vbak_kunnr
          kunnrwe  = ls_row-vbrk_kunwe
          augru    = ls_row-vbak_augru
          bstnk    = ls_row-vbak_bstnk
          bstdk    = ls_row-vbak_bstdk
          vdatu    = ls_row-vbak_vdatu
          name1    = ls_row-kna1_name1
          name2    = ls_row-kna1_name2
          stras    = ls_row-kna1_stras
          strsuppl = ls_row-adrc_str_suppl1
          ort01    = ls_row-kna1_ort01
          regio    = ls_row-kna1_regio
          pstlz    = ls_row-kna1_pstlz
          smtpaddr = ls_row-adr6_smtp_addr
          posnr    = ls_row-vbap_posnr
          matnr    = ls_row-vbap_matnr
          kwmeng   = ls_row-vbap_kwmeng
          vrkme    = ls_row-vbap_meins
          charg    = ls_row-vbap_charg
          werks    = ls_row-vbap_werks
      ) TO lt_create.
      lv_counter = lv_counter + 1.
    ENDLOOP.

    "── EML: create staging records ──────────────────────────────────────────
    MODIFY ENTITIES OF zi_sd_stg_orders IN LOCAL MODE
      ENTITY orders
        CREATE FIELDS (
            auart vkorg vtweg spart vkbur
            kunnrag kunnrwe augru
            bstnk bstdk vdatu
            name1 name2 stras strsuppl
            ort01 regio pstlz smtpaddr
            posnr matnr kwmeng vrkme charg werks
        )
        WITH lt_create
      MAPPED   DATA(lv_mapped)
      FAILED   DATA(lv_failed_create)
      REPORTED DATA(lv_reported_create).

    IF lv_failed_create-orders IS NOT INITIAL.
      APPEND LINES OF lv_failed_create-orders   TO failed-orders.
      APPEND LINES OF lv_reported_create-orders TO reported-orders.
      RETURN.
    ENDIF.

    "── Read back created instances for the action result ────────────────────
    READ ENTITIES OF zi_sd_stg_orders IN LOCAL MODE
      ENTITY orders
        ALL FIELDS
        WITH CORRESPONDING #( lv_mapped-orders )
      RESULT lt_created
      FAILED DATA(lv_read_failed).

    "── Populate result (return type = ZC_SD_STG_ORDERS / $self) ─────────────
    result = VALUE #(
        FOR ls_created IN lt_created
        ( %cid       = ls_key-%cid
          %param     = VALUE #( BASE CORRESPONDING #( ls_created )
                                Status = COND #( WHEN ls_created-status IS INITIAL THEN 'N'
                                                 ELSE ls_created-status )
                                StatusMsg = COND #( WHEN ls_created-statusmsg IS INITIAL THEN 'New'
                                                 ELSE ls_created-statusmsg )
                                                 ) ) ).

  ENDMETHOD.

  METHOD staging_data.

    DATA(lv_rectype) = keys[ 1 ]-%param-rectype.
    CASE lv_rectype.
      WHEN 'I'.  " New or Error or Modified records
        SELECT stg_guid FROM zsd_stg_orders
            WHERE status IN ( 'N','E','M' ) AND
            created_by = @sy-uname
            INTO TABLE @DATA(lt_keys).
      WHEN 'S'.  " Successful records
        SELECT stg_guid FROM zsd_stg_orders
            WHERE status IN ( 'S' ) AND
            created_by = @sy-uname
            INTO TABLE @lt_keys.
      WHEN OTHERS.
        RETURN.
    ENDCASE.

    READ ENTITIES OF zi_sd_stg_orders IN LOCAL MODE
        ENTITY orders
          ALL FIELDS
          WITH VALUE #( FOR ls_key IN lt_keys
                        ( stgguid = ls_key-stg_guid ) )
        RESULT DATA(lt_orders)
        FAILED DATA(lv_failed).

    DATA(lv_cid) = keys[ 1 ]-%cid.

    result = VALUE #(
        FOR ls_order IN lt_orders
        ( %cid   = lv_cid
          %param = CORRESPONDING #( ls_order ) ) ).

  ENDMETHOD.

  METHOD process_staging.

    " ── Mapping: task name → stg_guid ────────────────────────────────────
    DATA: BEGIN OF ls_map,
            taskname TYPE char32,
            stg_guid TYPE zsd_stg_orders-stg_guid,
          END OF ls_map.
    DATA lt_map LIKE TABLE OF ls_map.

    SELECT stg_guid, auart, vkorg, vtweg, spart, vkbur,
           kunnr_ag, kunnr_we, augru, bstnk, bstdk, vdatu,
           name1, name2, stras, str_suppl, ort01, regio, pstlz,
           smtp_addr, posnr, matnr, kwmeng, vrkme, charg, werks
      FROM zsd_stg_orders
      WHERE status IN ( 'N','E', 'M' ) AND
              created_by = @sy-uname
      INTO TABLE @DATA(lt_new_orders).

    IF lt_new_orders IS INITIAL.
      RETURN.
    ENDIF.

    CLEAR: gv_tasks_done, gt_task_results.

    DATA lt_update     TYPE TABLE FOR UPDATE zi_sd_stg_orders\\orders.
    DATA lv_timestamp  TYPE timestampl.
    DATA lv_batch_done TYPE i.                  " tasks fired in current batch
    CONSTANTS lc_max_parallel TYPE i VALUE 5.  " ← max concurrent tasks

    GET TIME STAMP FIELD lv_timestamp.

    LOOP AT lt_new_orders INTO DATA(ls_order).

      DATA(lv_tabix) = sy-tabix.
      "DATA(lv_task)  = ls_order-stg_guid.      " unique task name per order

      DATA(lv_task) = |TASK_{ sy-tabix }|.  " prefix to ensure it fits char10
      APPEND VALUE #( taskname = lv_task  stg_guid = ls_order-stg_guid ) TO lt_map.
      " ── Header ──────────────────────────────────────────────────────────
      DATA(ls_header) = VALUE bapisdhd1(
          doc_type   = ls_order-auart
          sales_org  = ls_order-vkorg
          distr_chan  = ls_order-vtweg
          division   = ls_order-spart
          sales_off  = ls_order-vkbur
          purch_no_c = ls_order-bstnk
          purch_date = ls_order-bstdk
          req_date_h = ls_order-vdatu
      ).

      " ── Partners ────────────────────────────────────────────────────────
      DATA lt_partners TYPE TABLE OF bapiparnr.
      CLEAR lt_partners.
      IF ls_order-kunnr_ag IS NOT INITIAL.
        APPEND VALUE #( partn_role = 'AG' partn_numb = |{ ls_order-kunnr_ag ALPHA = IN }| ) TO lt_partners.
      ENDIF.
      IF ls_order-kunnr_we IS NOT INITIAL.
        APPEND VALUE #( partn_role = 'WE' partn_numb = |{ ls_order-kunnr_we ALPHA = IN }| ) TO lt_partners.
      ENDIF.

      " ── Item ────────────────────────────────────────────────────────────
      DATA lt_items TYPE TABLE OF bapisditm.
      CLEAR lt_items.
      APPEND VALUE #(
          itm_number = ls_order-posnr
          material   = ls_order-matnr
          target_qty = ls_order-kwmeng
          sales_unit = ls_order-vrkme
          batch      = ls_order-charg
          plant      = ls_order-werks
      ) TO lt_items.

      " ── Schedule line ───────────────────────────────────────────────────
      DATA lt_schedules TYPE TABLE OF bapischdl.
      CLEAR lt_schedules.
      APPEND VALUE #(
          itm_number = ls_order-posnr
          req_date   = ls_order-vdatu
          req_qty    = ls_order-kwmeng
      ) TO lt_schedules.

      " ── Fire async task ─────────────────────────────────────────────────
      CALL FUNCTION 'Z_SD_CREATE_SALESORDER'
        STARTING NEW TASK lv_task
        CALLING me->on_bapi_return ON END OF TASK
        EXPORTING
          i_header    = ls_header
        TABLES
          t_items     = lt_items
          t_partners  = lt_partners
          t_schedules = lt_schedules.

      lv_batch_done = lv_batch_done + 1.

      " ── If batch of 5 reached, wait for all in this batch to finish ─────
      IF lv_batch_done >= lc_max_parallel.

        DATA(lv_expected) = gv_tasks_done + lv_batch_done.
        WAIT FOR ASYNCHRONOUS TASKS UNTIL gv_tasks_done >= lv_expected
          UP TO 300 SECONDS.

        CLEAR lv_batch_done.   " reset counter for next batch

      ENDIF.

    ENDLOOP.

    " ── Wait for any remaining tasks in the last (partial) batch ────────────
    IF lv_batch_done > 0.
      DATA(lv_total_tasks) = lines( lt_new_orders ).
      WAIT FOR ASYNCHRONOUS TASKS UNTIL gv_tasks_done >= lv_total_tasks
        UP TO 300 SECONDS.
    ENDIF.

    " ── Build EML update from collected results ──────────────────────────────
    GET TIME STAMP FIELD lv_timestamp.

    LOOP AT lt_new_orders INTO ls_order.

      DATA(lv_task_name) = VALUE char32( lt_map[ stg_guid = ls_order-stg_guid ]-taskname OPTIONAL ).
      READ TABLE gt_task_results WITH KEY task = lv_task_name INTO DATA(ls_result).

*      READ TABLE gt_task_results
*        WITH KEY stgguid = ls_order-stg_guid
*        INTO DATA(ls_result).

      IF sy-subrc = 0 AND ls_result-success = abap_true.
        APPEND VALUE #(
            stgguid                = ls_order-stg_guid
            status                 = 'S'
            statusmsg              = |SO { ls_result-vbeln } created|
            lastchangedat          = lv_timestamp
            lastchangedby          = sy-uname "cl_abap_context_info=>get_user_alias( )
            %control-status        = if_abap_behv=>mk-on
            %control-statusmsg     = if_abap_behv=>mk-on
            %control-lastchangedat = if_abap_behv=>mk-on
            %control-lastchangedby = if_abap_behv=>mk-on
        ) TO lt_update.
      ELSE.
        APPEND VALUE #(
            stgguid                = ls_order-stg_guid
            status                 = 'E'
            statusmsg              = COND #( WHEN ls_result-message IS NOT INITIAL
                                             THEN ls_result-message
                                             ELSE 'BAPI call failed or timed out' )
            lastchangedat          = lv_timestamp
            lastchangedby          = sy-uname "cl_abap_context_info=>get_user_alias( )
            %control-status        = if_abap_behv=>mk-on
            %control-statusmsg     = if_abap_behv=>mk-on
            %control-lastchangedat = if_abap_behv=>mk-on
            %control-lastchangedby = if_abap_behv=>mk-on
        ) TO lt_update.
      ENDIF.

    ENDLOOP.

    IF lt_update IS NOT INITIAL.
      MODIFY ENTITIES OF zi_sd_stg_orders IN LOCAL MODE
        ENTITY orders
          UPDATE FIELDS ( status statusmsg lastchangedat lastchangedby )
          WITH lt_update
        REPORTED DATA(lv_reported).

      "── Read back updated records to return as result ────────────────────
      READ ENTITIES OF zi_sd_stg_orders IN LOCAL MODE
        ENTITY orders
          ALL FIELDS
          WITH VALUE #( FOR ls_upd IN lt_update
                        ( stgguid = ls_upd-stgguid ) )
        RESULT DATA(lt_result_orders)
        FAILED DATA(lv_read_failed).

      "── Populate result using CID from keys[ 1 ] (static action) ─────────
      DATA(lv_cid) = keys[ 1 ]-%cid.

      result = VALUE #(
          FOR ls_res IN lt_result_orders
          ( %cid   = lv_cid
            %param = CORRESPONDING #( ls_res ) ) ).
    ENDIF.

  ENDMETHOD.


  METHOD on_bapi_return.

    DATA lt_return TYPE TABLE OF bapiret2.
    DATA lv_vbeln  TYPE vbeln_va.

    RECEIVE RESULTS FROM FUNCTION 'Z_SD_CREATE_SALESORDER'
      IMPORTING
        e_vbeln  = lv_vbeln
      TABLES
        t_return = lt_return.

    DATA(lv_success) = abap_true.
    DATA(lv_message) = ||.

    LOOP AT lt_return INTO DATA(ls_ret)
      WHERE type CA 'EAX'.
      lv_success = abap_false.
      lv_message = |{ ls_ret-message }|.
      EXIT.
    ENDLOOP.

    " No COMMIT/ROLLBACK here - already done inside Z_SD_CREATE_SALESORDER

    APPEND VALUE #(
        task = p_task
        vbeln   = lv_vbeln
        success = lv_success
        message = lv_message
    ) TO gt_task_results.

    ADD 1 TO gv_tasks_done.


  ENDMETHOD.

  METHOD delete_staging.

    "Get all Guids for deletion from DB Table ZSD_STG_ORDERS
    SELECT stg_guid FROM zsd_stg_orders
      WHERE status <> ' ' "AND created_by = @sy-uname
      INTO TABLE @DATA(lt_keys).

    IF lt_keys IS NOT INITIAL.

      DATA lt_delete TYPE TABLE FOR DELETE zi_sd_stg_orders\\orders.

      lt_delete = VALUE #( FOR ls_key IN lt_keys
                           ( %key-stgguid = ls_key-stg_guid ) ).

      MODIFY ENTITIES OF zi_sd_stg_orders IN LOCAL MODE
        ENTITY orders
          DELETE FROM
          lt_delete
        FAILED   DATA(lv_failed)
        REPORTED DATA(lv_reported).

      "return a success message after deletion
      APPEND VALUE #(
          %msg = new_message_with_text(
                      severity = if_abap_behv_message=>severity-success
                      text     = 'Staging records deleted successfully' )
      ) TO reported-orders.
    ELSE.
      " No records to delete, return message
      APPEND VALUE #(
          %msg = new_message_with_text(
                      severity = if_abap_behv_message=>severity-information
                      text     = 'No staging records to delete' )
      ) TO reported-orders.
    ENDIF.
  ENDMETHOD.

METHOD process_staging.

  DATA: BEGIN OF ls_map,
          taskname TYPE char32,
          stg_guid TYPE zsd_stg_orders-stg_guid,
        END OF ls_map.
  DATA lt_map LIKE TABLE OF ls_map.

  "── Fetch pending headers ────────────────────────────────────────────────────
  SELECT stg_guid, auart, vkorg, vtweg, spart, vkbur,
         kunnr_ag, kunnr_we, augru, bstnk, bstdk, vdatu,
         name1, name2, stras, str_suppl, ort01, regio, pstlz, smtp_addr
    FROM zsd_stg_orders
    WHERE status IN ( 'N','M' )
      AND created_by = @sy-uname
    INTO TABLE @DATA(lt_headers).

  IF lt_headers IS INITIAL.
    RETURN.
  ENDIF.

  "── Fetch ALL items for the above headers in one SELECT ─────────────────────
  SELECT stg_guid, posnr, matnr, kwmeng, vrkme, charg, werks
    FROM zsd_stg_orders_i
    FOR ALL ENTRIES IN @lt_headers
    WHERE stg_guid = @lt_headers-stg_guid
    INTO TABLE @DATA(lt_all_items).

  CLEAR: gv_tasks_done, gt_task_results.

  DATA lt_update     TYPE TABLE FOR UPDATE zi_sd_stg_orders\\orders.
  DATA lv_timestamp  TYPE timestampl.
  DATA lv_batch_done TYPE i.
  CONSTANTS lc_max_parallel TYPE i VALUE 5.

  GET TIME STAMP FIELD lv_timestamp.

  LOOP AT lt_headers INTO DATA(ls_hdr).

    DATA(lv_task) = |TASK_{ sy-tabix }|.
    APPEND VALUE #( taskname = lv_task  stg_guid = ls_hdr-stg_guid ) TO lt_map.

    "── Header ──────────────────────────────────────────────────────────────
    DATA(ls_header) = VALUE bapisdhd1(
        doc_type   = ls_hdr-auart
        sales_org  = ls_hdr-vkorg
        distr_chan  = ls_hdr-vtweg
        division   = ls_hdr-spart
        sales_off  = ls_hdr-vkbur
        purch_no_c = ls_hdr-bstnk
        purch_date = ls_hdr-bstdk
        req_date_h = ls_hdr-vdatu
    ).

    "── Partners ────────────────────────────────────────────────────────────
    DATA lt_partners TYPE TABLE OF bapiparnr.
    CLEAR lt_partners.
    IF ls_hdr-kunnr_ag IS NOT INITIAL.
      APPEND VALUE #( partn_role = 'AG' partn_numb = |{ ls_hdr-kunnr_ag ALPHA = IN }| ) TO lt_partners.
    ENDIF.
    IF ls_hdr-kunnr_we IS NOT INITIAL.
      APPEND VALUE #( partn_role = 'WE' partn_numb = |{ ls_hdr-kunnr_we ALPHA = IN }| addr_link = '$1' ) TO lt_partners.
    ENDIF.

    "── ALL items for this header ────────────────────────────────────────────
    DATA lt_items     TYPE TABLE OF bapisditm.
    DATA lt_schedules TYPE TABLE OF bapischdl.
    CLEAR: lt_items, lt_schedules.

    LOOP AT lt_all_items INTO DATA(ls_item)
        WHERE stg_guid = ls_hdr-stg_guid.

      APPEND VALUE #(
          itm_number = ls_item-posnr
          material   = ls_item-matnr
          target_qty = ls_item-kwmeng
          sales_unit = ls_item-vrkme
          batch      = ls_item-charg
          plant      = ls_item-werks
      ) TO lt_items.

      APPEND VALUE #(
          itm_number = ls_item-posnr
          req_date   = ls_hdr-vdatu
          req_qty    = ls_item-kwmeng
      ) TO lt_schedules.

    ENDLOOP.

    "── Partner address ─────────────────────────────────────────────────────
    DATA lt_partneraddress TYPE TABLE OF bapiaddr1.
    CLEAR lt_partneraddress.
    APPEND VALUE #(
        addr_no    = '$1'
        name       = ls_hdr-name1
        name_2     = ls_hdr-name2
        street     = ls_hdr-stras
        str_suppl1 = ls_hdr-str_suppl
        city       = ls_hdr-ort01
        region     = ls_hdr-regio
        postl_cod1 = ls_hdr-pstlz
        e_mail     = ls_hdr-smtp_addr
    ) TO lt_partneraddress.

    "── Fire async task (one per header / PO Number) ────────────────────────
    CALL FUNCTION 'Z_SD_CREATE_SALESORDER'
      STARTING NEW TASK lv_task
      CALLING me->on_bapi_return ON END OF TASK
      EXPORTING
        i_header    = ls_header
      TABLES
        t_items          = lt_items
        t_partners       = lt_partners
        t_schedules      = lt_schedules
        t_partneraddress = lt_partneraddress.

    lv_batch_done = lv_batch_done + 1.

    IF lv_batch_done >= lc_max_parallel.
      DATA(lv_expected) = gv_tasks_done + lv_batch_done.
      WAIT FOR ASYNCHRONOUS TASKS UNTIL gv_tasks_done >= lv_expected
        UP TO 300 SECONDS.
      CLEAR lv_batch_done.
    ENDIF.

  ENDLOOP.

  IF lv_batch_done > 0.
    DATA(lv_total_tasks) = lines( lt_headers ).
    WAIT FOR ASYNCHRONOUS TASKS UNTIL gv_tasks_done >= lv_total_tasks
      UP TO 300 SECONDS.
  ENDIF.

  "── Build EML update from results ────────────────────────────────────────────
  GET TIME STAMP FIELD lv_timestamp.

  LOOP AT lt_headers INTO ls_hdr.
    DATA(lv_task_name) = VALUE char32(
        lt_map[ stg_guid = ls_hdr-stg_guid ]-taskname OPTIONAL ).
    READ TABLE gt_task_results WITH KEY task = lv_task_name INTO DATA(ls_result).

    IF sy-subrc = 0 AND ls_result-success = abap_true.
      APPEND VALUE #(
          stgguid                = ls_hdr-stg_guid
          status                 = 'S'
          statusmsg              = |SO { ls_result-vbeln } created|
          vbeln                  = ls_result-vbeln
          lastchangedat          = lv_timestamp
          lastchangedby          = sy-uname
          %control-status        = if_abap_behv=>mk-on
          %control-statusmsg     = if_abap_behv=>mk-on
          %control-vbeln         = if_abap_behv=>mk-on
          %control-lastchangedat = if_abap_behv=>mk-on
          %control-lastchangedby = if_abap_behv=>mk-on
      ) TO lt_update.
    ELSE.
      APPEND VALUE #(
          stgguid                = ls_hdr-stg_guid
          status                 = 'E'
          statusmsg              = COND #( WHEN ls_result-message IS NOT INITIAL
                                           THEN ls_result-message
                                           ELSE 'BAPI call failed or timed out' )
          lastchangedat          = lv_timestamp
          lastchangedby          = sy-uname
          %control-status        = if_abap_behv=>mk-on
          %control-statusmsg     = if_abap_behv=>mk-on
          %control-lastchangedat = if_abap_behv=>mk-on
          %control-lastchangedby = if_abap_behv=>mk-on
      ) TO lt_update.
    ENDIF.
  ENDLOOP.

  IF lt_update IS NOT INITIAL.
    MODIFY ENTITIES OF zi_sd_stg_orders IN LOCAL MODE
      ENTITY orders
        UPDATE FIELDS ( status statusmsg vbeln lastchangedat lastchangedby )
        WITH lt_update
      REPORTED DATA(lv_reported).

    READ ENTITIES OF zi_sd_stg_orders IN LOCAL MODE
      ENTITY orders ALL FIELDS
      WITH VALUE #( FOR ls_upd IN lt_update ( stgguid = ls_upd-stgguid ) )
      RESULT DATA(lt_result_orders)
      FAILED DATA(lv_read_failed).

    DATA(lv_cid) = keys[ 1 ]-%cid.
    result = VALUE #(
        FOR ls_res IN lt_result_orders
        ( %cid   = lv_cid
          %param = CORRESPONDING #( ls_res ) ) ).
  ENDIF.

ENDMETHOD.
